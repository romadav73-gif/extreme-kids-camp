  setInterval(() => {
    tickTimer();
    const today = currentDate();
    if (date === addDays(today,-1) && clockFmt.format(new Date()).startsWith('00:')) { date = today; ensure(date); render(); }
  }, 1000);

  async function boot() {
    ensure(date); render(); checkHealth();
    if (!session?.token) return;
    $('#auth').classList.add('hidden'); $('#app').classList.remove('hidden'); $('#mobileNav').classList.remove('hidden');
    await pullState(true); render(); startPolling();
  }

  boot();
